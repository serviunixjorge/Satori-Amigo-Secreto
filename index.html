async function syncFromCloud(userTriggered) { if (isViewingParchment) return; const btn = document.getElementById('syncBtn'); if (userTriggered) btn.innerText = "⏳..."; try { // Corrección: Se elimina el '?t=Date.now()' de la URL. // Se previene el caché utilizando los headers y configuraciones nativas de fetch. const res = await fetch(DB_URL, { cache: 'no-store', headers: { 'Cache-Control': 'no-cache', 'Pragma': 'no-cache' } }); if (res.ok) { const cloudText = await res.text(); if (cloudText) { // Evita errores si la base de datos devuelve texto vacío const cloudMap = JSON.parse(cloudText); if (cloudMap && typeof cloudMap === 'object') { burnedMap = cloudMap; localStorage.setItem('burned_cache_kvdb_g19', JSON.stringify(burnedMap)); renderGrid(); } } } } catch(e) { console.warn("Error leyendo de KVdb:", e); } finally { if (userTriggered) { setTimeout(() => { btn.innerText = "🔄 Sincronizar"; }, 300); } } } async function saveToCloud(newMap) { try { await fetch(DB_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' // Corrección: Especificar el tipo de contenido }, body: JSON.stringify(newMap) }); } catch(e){ console.error("Error guardando en KVdb:", e); } } async function broadcastBurn(name) { // Mejora: Obtener el estado más reciente de la nube antes de guardar. // Esto evita que si dos personas interactúan al mismo tiempo, uno borre el progreso del otro. try { const res = await fetch(DB_URL, { cache: 'no-store', headers: { 'Cache-Control': 'no-cache' } }); if (res.ok) { const text = await res.text(); if (text) { const cloudMap = JSON.parse(text); burnedMap = { ...burnedMap, ...cloudMap }; // Fusiona los datos locales con la nube } } } catch(e) {} burnedMap[name] = true; localStorage.setItem('burned_cache_kvdb_g19', JSON.stringify(burnedMap)); renderGrid(); await saveToCloud(burnedMap); }
